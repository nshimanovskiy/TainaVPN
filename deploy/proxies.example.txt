# Кладите прокси в файлы *.txt в папке proxies/ (рядом с docker-compose.yml).
# Одна строка — один прокси. Строки с # — комментарии.
#
# user:pass@host:port            SOCKS5 (по умолчанию)
# socks5://user:pass@host:port   SOCKS5
# http://user:pass@host:port     HTTP-прокси
# host:port:user:pass            SOCKS5, другая запись
#
# Каждому пользователю приложения выдаётся свой прокси (сначала свободные,
# потом наименее загруженные). Нерабочие прокси автоматически пропускаются.
